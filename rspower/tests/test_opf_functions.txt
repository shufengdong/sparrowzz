// Polynomial and piecewise-linear generator costs.
cost = [
    [2, 0, 0, 3, 2, 3, 4, 0, 0, 0],
    [1, 0, 0, 3, 0, 0, 10, 20, 20, 60]
];
assert_relative_eq(polycost(select(cost, [0]), [2], 0), [18], 1e-12);
assert_relative_eq(polycost(select(cost, [0]), [2], 1), [11], 1e-12);
assert_relative_eq(polycost(select(cost, [0]), [2], 2), [4], 1e-12);
assert_relative_eq(polycost(select(cost, [0]), [[2]], 0), [[18]], 1e-12);
assert_relative_eq(totcost(cost, [[2, 3, 4], [-5, 15, 25]]),
    [[18, 31, 48], [-10, 40, 80]], 1e-12);

[f_cost, df_cost, d2f_cost] = opf_gen_cost_fcn(
    [[0.02]], 100, select(cost, [0]), []
);
assert_relative_eq(f_cost, 18, 1e-12);
assert_relative_eq(df_cost, [[1100]], 1e-12);
assert_relative_eq(full(d2f_cost), [[40000]], 1e-12);

// Two-bus power balance with an exactly balanced real-power operating point.
bus = [
    [1, 3, 0, 0, 0, 0, 1, 1.0, 0, 100, 1, 1.1, 0.9],
    [2, 1, 9, 0, 0, 0, 1, 0.9, 0, 100, 1, 1.1, 0.9]
];
gen = [[1, 10, 0, 100, -100, 1, 100, 1, 100, 0]];
Ybus = sparse([0, 0, 1, 1], [0, 1, 0, 1], [1, -1, -1, 1], 2, 2);
[g_balance, dg_balance] = opf_power_balance_fcn(
    [[0], [0]], [[1], [0.9]], [[0.1]], [[0]], 100, bus, gen, Ybus, 0
);
assert_relative_eq(g_balance, [[0], [0], [0], [0]], 1e-12);
assert_relative_eq(full(dg_balance), [
    [0, 0, 1.1, -1.0, -1, 0],
    [0, 0, -0.9, 0.8, 0, 0],
    [-0.9, 0.9, 0, 0, 0, -1],
    [0.9, -0.9, 0, 0, 0, 0]
], 1e-12);

// Cartesian voltage constraints and their Jacobians.
Vr = [[1.0], [0.8]];
Vi = [[0.0], [0.6]];
[vlim, dvlim] = opf_vlim_fcn(Vr, Vi, bus, [0, 1]);
assert_relative_eq(vlim, [[-0.19], [-0.19], [-0.21], [-0.21]], 1e-12);
assert_relative_eq(full(dvlim), [
    [-2, 0, 0, 0], [0, -1.6, 0, -1.2],
    [2, 0, 0, 0], [0, 1.6, 0, 1.2]
], 1e-12);

[veq, dveq] = opf_veq_fcn(Vr, Vi, bus, [1]);
assert_relative_eq(veq, [[-0.21]], 1e-12);
assert_relative_eq(full(dveq), [[0, 1.6, 0, 1.2]], 1e-12);

[vref, dvref] = opf_vref_fcn(Vr, Vi, bus, [0]);
assert_relative_eq(vref, [[0]], 1e-12);
assert_relative_eq(full(dvref), [[0, 0, 1, 0]], 1e-12);

Aang = sparse([0], [0], [1], 1, 2) - sparse([0], [1], [1], 1, 2);
[vadif, dvadif] = opf_branch_ang_fcn(Vr, Vi, Aang, [-1], [1]);
assert_relative_eq(vadif, [[-0.35649889120671563], [-1.6435011087932844]], 1e-12);
assert_relative_eq(full(dvadif), [[0, -0.6, -1, 0.8], [0, 0.6, 1, -0.8]], 1e-12);

// One lossless branch exercises S, P, P^2 and I limits in both coordinates.
branch = [[1, 2, 0, 1, 0, 200, 200, 200, 0, 0, 1, -360, 360]];
[Ybus_branch, Yf, Yt] = makeYbus(100, bus, branch);
[h_s, dh_s] = opf_branch_flow_fcn(Vr, Vi, 100, branch, Yf, Yt, [0], 1, 0);
assert_relative_eq(h_s, [[-3.6], [-3.6]], 1e-12);
assert_relative_eq(full(dh_s), [
    [1.2, -0.4, -1.2, 1.2],
    [0.4, 0.24, -1.2, 1.68]
], 1e-12);

[h_p, dh_p] = opf_branch_flow_fcn(Vr, Vi, 100, branch, Yf, Yt, [0], 1, 1);
assert_relative_eq(h_p, [[-2.6], [-1.4]], 1e-12);
assert_relative_eq(full(dh_p), [
    [-0.6, 0, 0.8, -1],
    [0.6, 0, -0.8, 1]
], 1e-12);

[h_p2, dh_p2] = opf_branch_flow_fcn(Vr, Vi, 100, branch, Yf, Yt, [0], 1, 2);
assert_relative_eq(h_p2, [[-3.64], [-3.64]], 1e-12);
assert_relative_eq(full(dh_p2), [
    [0.72, 0, -0.96, 1.2],
    [0.72, 0, -0.96, 1.2]
], 1e-12);

[h_i, dh_i] = opf_branch_flow_fcn(Vr, Vi, 100, branch, Yf, Yt, [0], 1, 3);
assert_relative_eq(h_i, [[-3.6], [-3.6]], 1e-12);
assert_relative_eq(full(dh_i), [
    [0.4, -0.4, -1.2, 1.2],
    [0.4, -0.4, -1.2, 1.2]
], 1e-12);

Va = angle(Vr + 1j * Vi);
Vm = abs(Vr + 1j * Vi);
[h_s_polar, dh_s_polar] = opf_branch_flow_fcn(Va, Vm, 100, branch, Yf, Yt, [0], 0, 0);
[h_i_polar, dh_i_polar] = opf_branch_flow_fcn(Va, Vm, 100, branch, Yf, Yt, [0], 0, 3);
assert_relative_eq(h_s_polar, h_s, 1e-12);
assert_relative_eq(h_i_polar, h_i, 1e-12);
assert_eq(size(dh_s_polar, 1), 4);
assert_eq(size(dh_i_polar, 1), 4);
