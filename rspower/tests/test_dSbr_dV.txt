#include ../data/case3bus_P6_6.txt
#include ../lib/idx_gen.txt
#include ../lib/idx_bus.txt
#include ../lib/idx_brch.txt
#include ../lib/makeYbus.txt
#include ../lib/dSbr_dV.txt

V = [[1], [1.02], [1.02]];
[Ybus, Yf, Yt] = makeYbus(baseMVA, bus, branch);
vcart = 0;

[dSf_dV1, dSf_dV2, dSt_dV1, dSt_dV2, Sf, St] = dSbr_dV(branch, Yf, Yt, V, vcart);

assert_relative_eq(full(dSf_dV1), [
    [c(10.0990, -1.0099), c(-10.0990, 1.0099), c(0,0)],
    [c(8.1600, -4.0800), c(0, 0), c(-8.1600, 4.0800)],
    [c(0, 0), c(8.3232, -4.1616), c(-8.3232, 4.1616)]
],0.001);

assert_relative_eq(full(dSf_dV2), [
    [c(0.9703, 9.6530), c(-0.9901, -9.9010), c(0,0)],
    [c(3.9200, 7.8150), c(0, 0), c(-4.0000, -8.0000)],
    [c(0, 0), c(4.0800, 8.1345), c(-4.0800, -8.1600)]
],0.001);

assert_relative_eq(full(dSt_dV1), [
    [c(-10.0990, 1.0099), c(10.0990, -1.0099), c(0,0)],
    [c(-8.1600, 4.0800), c(0, 0), c(8.1600, -4.0800)],
    [c(0, 0), c(-8.3232, 4.1616), c(8.3232, -4.1616)]
],0.001);

assert_relative_eq(full(dSt_dV2), [
    [c(-1.0099, -10.0990), c(1.0297, 10.2460), c(0,0)],
    [c(-4.0800, -8.1600), c(0, 0), c(4.1600, 8.2945)],
    [c(0, 0), c(-4.0800, -8.1600), c(4.0800, 8.1345)]
],0.001);

assert_relative_eq(Sf, [
    [c(-0.0198, -0.2230)],
    [c(-0.0800, -0.1725)],
    [c(-4.4391e-16, -0.0130)]
],0.001);

assert_relative_eq(St, [
    [c(0.0202,0.1760)],
    [c(0.0816,0.1502)],
    [c(4.4391e-16, -0.0130)]
],0.001);

[dSf_dVa_sparse, dSf_dVm_sparse, dSt_dVa_sparse, dSt_dVm_sparse, Sf_sparse, St_sparse] = dSbr_dV(branch, Yf, Yt, V, 0);
[dSf_dVa_dense, dSf_dVm_dense, dSt_dVa_dense, dSt_dVm_dense, Sf_dense, St_dense] = dSbr_dV(branch, full(Yf), full(Yt), V, 0);

assert_relative_eq(full(dSf_dVa_sparse), dSf_dVa_dense, 0.0000001);
assert_relative_eq(full(dSf_dVm_sparse), dSf_dVm_dense, 0.0000001);
assert_relative_eq(full(dSt_dVa_sparse), dSt_dVa_dense, 0.0000001);
assert_relative_eq(full(dSt_dVm_sparse), dSt_dVm_dense, 0.0000001);
assert_relative_eq(Sf_sparse, Sf_dense, 0.0000001);
assert_relative_eq(St_sparse, St_dense, 0.0000001);

[dSf_dVr_sparse, dSf_dVi_sparse, dSt_dVr_sparse, dSt_dVi_sparse, Sf_sparse, St_sparse] = dSbr_dV(branch, Yf, Yt, V, 1);
[dSf_dVr_dense, dSf_dVi_dense, dSt_dVr_dense, dSt_dVi_dense, Sf_dense, St_dense] = dSbr_dV(branch, full(Yf), full(Yt), V, 1);

assert_relative_eq(full(dSf_dVr_sparse), dSf_dVr_dense, 0.0000001);
assert_relative_eq(full(dSf_dVi_sparse), dSf_dVi_dense, 0.0000001);
assert_relative_eq(full(dSt_dVr_sparse), dSt_dVr_dense, 0.0000001);
assert_relative_eq(full(dSt_dVi_sparse), dSt_dVi_dense, 0.0000001);
assert_relative_eq(Sf_sparse, Sf_dense, 0.0000001);
assert_relative_eq(St_sparse, St_dense, 0.0000001);
