// The single-element MATLAB cell input is a numeric dispatch vector here.
fn opf_gen_cost_fcn(PQg, baseMVA, gencost, ig) {
    if is_empty(ig) { ig=find(slice(gencost,[0],[MODEL-1,MODEL])==POLYNOMIAL); }
    ng=length(PQg);
    if length(ig)==0 { return (0,zeros(ng,1),sparse([],[],[],ng,ng)); }
    if ndim(PQg)==1 { PQg=transpose(PQg); }
    xx=select(PQg,ig)*baseMVA;
    cost=select(gencost,ig);
    values=totcost(cost,xx); f=sum_all(values);
    df=zeros(ng,1);
    first=polycost(cost,xx,1); second=polycost(cost,xx,2);
    df=set(df,ig,baseMVA*first);
    d2f=sparse(ig,ig,baseMVA^2*second,ng,ng);
    return (f,df,d2f);
}
