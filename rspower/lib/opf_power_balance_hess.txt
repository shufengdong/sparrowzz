// OPF_POWER_BALANCE_HESS  Hessian of lambda.' * [Pbalance;Qbalance].
// Returns a sparse matrix for x=[V1;V2;Pg;Qg]. ng is the generator count;
// generation and the currently supported constant-power loads are linear.
// MATPOWER 7.1, Copyright (c) 1996-2017 PSERC, Ray Zimmerman,
// Carlos E. Murillo-Sanchez and Baljinnyam Sereeter.
// Covered by the 3-clause BSD License; see LICENSE-MATPOWER.
fn opf_power_balance_hess(V1,V2,ng,lambda,Ybus,vcart) {
    if vcart { V=V1+1j*V2; } else { V=V2.*exp(1j*V1); }
    nb=length(V);
    lamP=slice(lambda,[0,nb],[0]);
    lamQ=slice(lambda,[nb,2*nb],[0]);
    [P11,P12,P21,P22]=d2Sbus_dV2(Ybus,V,lamP,vcart);
    [Q11,Q12,Q21,Q22]=d2Sbus_dV2(Ybus,V,lamQ,vcart);
    voltage=real(vertcat(horzcat(P11,P12),horzcat(P21,P22)))
        +imag(vertcat(horzcat(Q11,Q12),horzcat(Q21,Q22)));
    hessian=vertcat(
        horzcat(voltage,sparse([],[],[],2*nb,2*ng)),
        sparse([],[],[],2*ng,2*nb+2*ng));
    return hessian;
}
