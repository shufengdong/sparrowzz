fn totcost(gencost, Pg) {
    ng=size(gencost,0); vector=ndim(Pg)==1;
    if ng==0 { return zeros(size(Pg)); }
    if vector { Pg=transpose(Pg); }
    columns=size(Pg,1);
    totalcost=zeros(ng,columns);
    ipwl=find(slice(gencost,[0],[MODEL-1,MODEL])==PW_LINEAR);
    ipol=find(slice(gencost,[0],[MODEL-1,MODEL])==POLYNOMIAL);

    // mkpp/ppval are expressed as the equivalent linear interpolation here.
    for row in 0..length(ipwl) {
        i=get(ipwl,row); ncost=get(gencost,i,NCOST-1);
        for column in 0..columns {
            p=get(Pg,i,column); value=0;
            if ncost==1 { value=get(gencost,i,COST); }
            else if ncost>1 {
                segment=0;
                if p>=get(gencost,i,COST-1+2*(ncost-1)) { segment=ncost-2; }
                else {
                    while segment<ncost-2 && p>get(gencost,i,COST-1+2*(segment+1)) { segment=segment+1; }
                }
                x1=get(gencost,i,COST-1+2*segment); y1=get(gencost,i,COST+2*segment);
                x2=get(gencost,i,COST-1+2*(segment+1)); y2=get(gencost,i,COST+2*(segment+1));
                value=y1+(p-x1)*(y2-y1)/(x2-x1);
            }
            totalcost=set(totalcost,[i*columns+column],value);
        }
    }
    if length(ipol)>0 {
        for i in 0..columns {
            values=polycost(select(gencost,ipol),select(Pg,ipol,[i]),0);
            totalcost=select_assign(totalcost,values,ipol,[i]);
        }
    }
    if vector { return get_multi(totalcost,range(0,ng)); }
    return totalcost;
}
