// OPF_POWER_BALANCE_FCN  Evaluates AC power balance constraints and gradients.
//
//   [G, DG] = OPF_POWER_BALANCE_FCN(V1, V2, PG, QG, BASEMVA, BUS, GEN,
//                                    YBUS, VCART)
//
//   For VCART = 0, V1 and V2 are VA and VM. For VCART = 1, they are VR and
//   VI. PG and QG are in per unit. The MATPOWER mpc and mpopt structures are
//   represented by explicit matrices and VCART in Sparrowzz.
//
//   MATPOWER
//   Copyright (c) 1996-2017, Power Systems Engineering Research Center (PSERC)
//   Covered by the 3-clause BSD License. See https://matpower.org.

fn opf_power_balance_fcn(V1, V2, Pg, Qg, baseMVA, bus, gen, Ybus, vcart) {
    if vcart {
        V = V1 + 1j * V2;
    } else {
        V = V2 .* exp(1j * V1);
    }

    nb = length(V);
    ng = length(Pg);

    // Put per-unit dispatch back into the MATPOWER generator matrix.
    opf_gen = assign(
        gen, horzcat(Pg, Qg) * baseMVA, [0], [PG - 1, QG]
    );
    [Sbus, neg_dSd_dVm] = makeSbus(baseMVA, bus, opf_gen);

    mis = V .* conj(Ybus * V) - Sbus;
    g = vertcat(real(mis), imag(mis));

    [dSbus_dV1, dSbus_dV2] = dSbus_dV(Ybus, V, vcart);
    if ~~vcart {
        // makeSbus returns the negative load derivative. It is currently zero
        // for the supported constant-power model, but keep MATPOWER's formula.
        dSbus_dV2 = dSbus_dV2 - neg_dSd_dVm;
    }

    gbus = slice(opf_gen, [0], [GEN_BUS - 1, GEN_BUS]) - 1;
    neg_Cg = sparse(gbus, range(0, ng), -1, nb, ng);
    zero_gen = sparse([], [], [], nb, ng);

    dg = vertcat(
        horzcat(real(dSbus_dV1), real(dSbus_dV2), neg_Cg, zero_gen),
        horzcat(imag(dSbus_dV1), imag(dSbus_dV2), zero_gen, neg_Cg)
    );
    return (g, dg);
}
