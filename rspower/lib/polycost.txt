// DER is explicit; indices are 0-based. Preserve the input vector shape.
fn polycost(gencost, Pg, der) {
    if sum(slice(gencost,[0],[MODEL-1,MODEL])==PW_LINEAR)>0 {
        error(`polycost: all costs must be polynomial`);
    }
    ng=length(Pg); vector=ndim(Pg)==1;
    if ng==0 { return zeros(size(Pg)); }
    if vector { Pg=transpose(Pg); }
    ncost=slice(gencost,[0],[NCOST-1,NCOST]);
    maxN=max(ncost); minN=min(ncost);

    // Form coefficient matrix: constant term, linear term, etc.
    c=zeros(ng,maxN);
    for n in minN..maxN+1 {
        k=find(ncost==n);
        if length(k)>0 {
            coefficients=select(gencost,k,range(COST+n-2,COST-2,-1));
            c=select_assign(c,coefficients,k,range(0,n));
        }
    }

    // Differentiate coefficient columns, as in MATPOWER polycost.
    for d in 0..der {
        if size(c,1)>=2 { c=slice(c,[0],[1,maxN-d]); }
        else { c=zeros(ng,1); break; }
        for k in 1..maxN-d-1 {
            c=assign(c,(k+1)*slice(c,[0],[k,k+1]),[0],[k,k+1]);
        }
    }

    if is_empty(c) { f=zeros(ng,1); }
    else {
        f=slice(c,[0],[0,1]);
        for k in 1..size(c,1) { f=f+slice(c,[0],[k,k+1]).*Pg.^k; }
    }
    if vector { return get_multi(f,range(0,ng)); }
    return f;
}
